cmake_minimum_required(VERSION 3.25)
project(pcu_mlx_c_api_evaluation VERSION 1.0.0 LANGUAGES CXX)

if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
    message(FATAL_ERROR "Fusion PCU MLX requires Apple silicon macOS")
endif()
find_package(MLX 0.32.3 EXACT CONFIG REQUIRED)
set(MLX_C_SAFE_SOURCE "" CACHE PATH "Explicit safety-patched official mlx-c source copy")
if(NOT EXISTS "${MLX_C_SAFE_SOURCE}/safety-patch-provenance.json")
    message(FATAL_ERROR "Run prepare_source.py on the audited official source first")
endif()
# Bounded subset avoids unrelated fast.cpp API drift in the pinned Homebrew binding.
set(c_sources array device stream error evaluation_matmul closure vector map compile version string)
set(c_paths)
foreach(name IN LISTS c_sources)
    list(APPEND c_paths "${MLX_C_SAFE_SOURCE}/mlx/c/${name}.cpp")
endforeach()
add_library(pcu_mlx_c_official_safe OBJECT ${c_paths})
target_compile_features(pcu_mlx_c_official_safe PRIVATE cxx_std_20)
target_include_directories(pcu_mlx_c_official_safe PRIVATE "${MLX_C_SAFE_SOURCE}")
target_link_libraries(pcu_mlx_c_official_safe PRIVATE mlx)
set_target_properties(pcu_mlx_c_official_safe PROPERTIES
    POSITION_INDEPENDENT_CODE YES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_library(pcu_mlx_c_api_evaluation SHARED bridge.cpp $<TARGET_OBJECTS:pcu_mlx_c_official_safe>)
target_compile_features(pcu_mlx_c_api_evaluation PRIVATE cxx_std_20)
target_include_directories(pcu_mlx_c_api_evaluation SYSTEM PRIVATE "${MLX_C_SAFE_SOURCE}")
target_link_libraries(pcu_mlx_c_api_evaluation PRIVATE mlx)
set_target_properties(pcu_mlx_c_api_evaluation PROPERTIES
    CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
    BUILD_RPATH "${MLX_LIBRARY_DIRS}")
target_compile_options(pcu_mlx_c_api_evaluation PRIVATE -Wall -Wextra -Wpedantic -Werror)
# Official C entry points/handler remain local to this owned comparison image.
set(exports "${CMAKE_CURRENT_BINARY_DIR}/evaluation.exports")
file(WRITE "${exports}" "_eval_abi\n_eval_version\n_eval_session_new\n_eval_session_free\n_eval_array_new\n_eval_array_free\n_eval_read\n_eval_matmul\n_eval_prepare\n_eval_compiled\n_eval_prepared_free\n_eval_traces\n")
target_link_options(pcu_mlx_c_api_evaluation PRIVATE "LINKER:-exported_symbols_list,${exports}")
